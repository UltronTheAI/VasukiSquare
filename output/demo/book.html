The following implementation outlines the state transition logic for distributed consensus.

class RaftNode:
    def __init__(self, node_id: str, peers: list[str]):
        self.node_id = node_id
        self.peers = peers
        self.current_term = 0
        self.voted_for = None
        self.log = []
        self.commit_index = 0

    async def start_election(self):
        self.current_term += 1
        self.voted_for = self.node_id
        votes = 1
        for peer in self.peers:
            if await self.request_vote(peer):
                votes += 1
        if votes > len(self.peers) // 2:
            self.become_leader()
